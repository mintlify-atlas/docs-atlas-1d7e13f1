# Checkout

Creates a checkout session for a customer to purchase a product with payment collection. Returns a checkout URL where the customer can complete their purchase.

## When to Use

Use this for new customers or when payment info is needed. For customers with existing payment methods, use `attach` instead.

## Authentication

Requires a valid API key passed in the `Authorization` header:

```bash
Authorization: Bearer YOUR_API_KEY
```

## Request Body

The checkout endpoint accepts the same parameters as the [attach](/api/core/attach) endpoint, with an additional `setup_payment` option.

<ParamField body="customer_id" type="string" required>
  The ID of the customer
  
  **Example:** `cus_123`
</ParamField>

<ParamField body="product_id" type="string">
  The ID of the product to purchase
  
  **Example:** `pro_plan`
  
  **Note:** Either `product_id` or `product_ids` must be provided
</ParamField>

<ParamField body="product_ids" type="string[]">
  Array of product IDs to purchase multiple products at once
  
  **Example:** `["pro_plan", "addon_storage"]`
</ParamField>

<ParamField body="entity_id" type="string">
  The ID of the entity (optional, for entity-scoped products)
  
  **Example:** `entity_123`
</ParamField>

<ParamField body="customer_data" type="object">
  Customer details to set when creating the customer
  
  <ParamField body="customer_data.name" type="string">
    Customer's name
  </ParamField>
  
  <ParamField body="customer_data.email" type="string">
    Customer's email address
  </ParamField>
  
  <ParamField body="customer_data.fingerprint" type="string">
    Unique identifier to detect duplicate customers and prevent free trial abuse
  </ParamField>
  
  <ParamField body="customer_data.metadata" type="object">
    Additional metadata for the customer
  </ParamField>
  
  <ParamField body="customer_data.stripe_id" type="string">
    Stripe customer ID if you already have one
  </ParamField>
</ParamField>

<ParamField body="entity_data" type="object">
  Additional entity properties. These will be used to create the entity if it doesn't exist.
</ParamField>

<ParamField body="options" type="array">
  Feature options for configuring specific features in the product
</ParamField>

<ParamField body="free_trial" type="boolean | object">
  Whether to start a free trial. Can be a boolean or a free trial configuration object.
  
  **Example:** `true` or `{ duration: 14, interval: "day" }`
</ParamField>

<ParamField body="success_url" type="string">
  URL to redirect to after successful payment
  
  **Example:** `https://example.com/success`
</ParamField>

<ParamField body="setup_payment" type="boolean">
  Whether to only setup payment method without immediately charging
  
  **Default:** `false`
</ParamField>

<ParamField body="checkout_session_params" type="object">
  Additional parameters to pass to the payment processor's checkout session (e.g., Stripe Checkout parameters)
</ParamField>

<ParamField body="reward" type="string | string[]">
  Reward ID(s) to grant upon successful checkout
</ParamField>

<ParamField body="invoice" type="boolean">
  Whether to create an invoice for this checkout
</ParamField>

<ParamField body="new_billing_subscription" type="boolean">
  Whether to create a new billing subscription
</ParamField>

<ParamField body="is_custom" type="boolean">
  Whether this is a custom product (requires `items` field)
</ParamField>

<ParamField body="items" type="array">
  Product items for custom products (only valid when `is_custom` is true)
</ParamField>

<ParamField body="metadata" type="object">
  Additional metadata to attach to the checkout session
</ParamField>

<ParamField body="billing_cycle_anchor" type="number">
  Unix timestamp for when the billing cycle should start
</ParamField>

## Response

<ResponseField name="url" type="string">
  URL to redirect the customer to for completing the checkout
</ResponseField>

<ResponseField name="customer_id" type="string" required>
  The ID of the customer
</ResponseField>

<ResponseField name="lines" type="array" required>
  Array of line items showing what will be charged
  
  <ResponseField name="lines[].description" type="string">
    Description of the line item
  </ResponseField>
  
  <ResponseField name="lines[].amount" type="number">
    Amount for this line item in cents
  </ResponseField>
  
  <ResponseField name="lines[].item" type="object">
    Product item details
  </ResponseField>
</ResponseField>

<ResponseField name="product" type="object">
  The product being purchased
</ResponseField>

<ResponseField name="current_product" type="object">
  The customer's current product (if upgrading/downgrading)
</ResponseField>

<ResponseField name="options" type="array">
  Selected feature options
</ResponseField>

<ResponseField name="total" type="number">
  Total amount to be charged in cents
</ResponseField>

<ResponseField name="currency" type="string">
  Currency code (e.g., "usd")
</ResponseField>

<ResponseField name="has_prorations" type="boolean">
  Whether this checkout includes prorations from a plan change
</ResponseField>

<ResponseField name="next_cycle" type="object">
  Information about the next billing cycle
  
  <ResponseField name="next_cycle.starts_at" type="number">
    Unix timestamp when the next cycle starts
  </ResponseField>
  
  <ResponseField name="next_cycle.total" type="number">
    Amount for the next cycle in cents
  </ResponseField>
</ResponseField>

```typescript TypeScript
interface CheckoutRequest {
  customer_id: string;
  product_id?: string;
  product_ids?: string[];
  entity_id?: string;
  customer_data?: {
    name?: string;
    email?: string;
    metadata?: Record<string, any>;
  };
  success_url?: string;
  setup_payment?: boolean;
  free_trial?: boolean;
  checkout_session_params?: Record<string, any>;
}

interface CheckoutResponse {
  url?: string;
  customer_id: string;
  lines: Array<{
    description: string;
    amount: number;
    item?: any;
  }>;
  product?: any;
  current_product?: any;
  total?: number;
  currency?: string;
  has_prorations?: boolean;
  next_cycle?: {
    starts_at: number;
    total: number;
  };
}

const response = await fetch('https://api.useautumn.com/checkout', {
  method: 'POST',
  headers: {
    'Authorization': 'Bearer YOUR_API_KEY',
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    customer_id: 'cus_123',
    product_id: 'pro_plan',
    customer_data: {
      name: 'John Doe',
      email: 'john@example.com'
    },
    success_url: 'https://example.com/success'
  } as CheckoutRequest)
});

const data: CheckoutResponse = await response.json();
```

## Use Cases

### Basic Checkout Flow

```typescript
const createCheckout = async (customerId: string, productId: string) => {
  const response = await fetch('https://api.useautumn.com/checkout', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${API_KEY}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      customer_id: customerId,
      product_id: productId,
      success_url: `${window.location.origin}/success`,
      checkout_session_params: {
        cancel_url: `${window.location.origin}/cancel`
      }
    })
  });
  
  const data = await response.json();
  
  // Redirect to checkout
  if (data.url) {
    window.location.href = data.url;
  }
  
  return data;
};
```

## Notes

- The checkout session URL is typically valid for 24 hours
- If the customer already has a payment method on file, consider using the `/attach` endpoint instead
- Use `success_url` to control where users are redirected after successful payment
- The `lines` array shows exactly what will be charged, including any prorations
- When upgrading/downgrading, prorations are calculated automatically based on time remaining in the billing period
- Custom checkout session parameters can be passed via `checkout_session_params` to customize the Stripe Checkout experience
- The response includes preview information (`total`, `lines`, `next_cycle`) that you can use to build a confirmation UI before redirecting
