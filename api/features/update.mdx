---
title: 'Update Feature'
openapi: 'POST /features/{feature_id}'
---

Update an existing feature's properties. All fields are optional.

## Path Parameters

<ParamField path="feature_id" type="string" required>
  The ID of the feature to update
</ParamField>

## Request Body

<ParamField path="id" type="string">
  The new ID for the feature. Must match the pattern: lowercase letters, numbers, hyphens, and underscores only.
</ParamField>

<ParamField path="name" type="string">
  The name of the feature
</ParamField>

<ParamField path="type" type="enum">
  The type of the feature: `single_use`, `continuous_use`, or `credit_system`
</ParamField>

<ParamField path="consumable" type="boolean">
  Whether the feature is consumable. Required when changing type to `single_use`.
</ParamField>

<ParamField path="display" type="object">
  Singular and plural display names for the feature
  
  <Expandable title="Display Object">
    <ParamField path="singular" type="string">
      Singular display name
    </ParamField>
    
    <ParamField path="plural" type="string">
      Plural display name
    </ParamField>
  </Expandable>
</ParamField>

<ParamField path="credit_schema" type="array">
  Credit schema for credit_system features
  
  <Expandable title="Credit Schema Item">
    <ParamField path="metered_feature_id" type="string">
      The ID of the single_use feature
    </ParamField>
    
    <ParamField path="credit_cost" type="number">
      The credit cost
    </ParamField>
  </Expandable>
</ParamField>

<ParamField path="event_names" type="array">
  Array of event names associated with this feature
</ParamField>

<ParamField path="archived" type="boolean">
  Whether the feature is archived. Archived features are hidden from the dashboard and list features endpoint.
</ParamField>

## Response

<ResponseField name="id" type="string">
  The ID of the updated feature
</ResponseField>

<ResponseField name="name" type="string">
  The name of the feature
</ResponseField>

<ResponseField name="type" type="enum">
  The type of the feature
</ResponseField>

<ResponseField name="consumable" type="boolean">
  Whether the feature is consumable
</ResponseField>

<ResponseField name="event_names" type="array">
  Array of event names
</ResponseField>

<ResponseField name="credit_schema" type="array">
  Credit schema mapping
</ResponseField>

<ResponseField name="display" type="object">
  Display names
</ResponseField>

<ResponseField name="archived" type="boolean">
  Whether the feature is archived
</ResponseField>

## Example

```json Request
{
  "name": "API Requests",
  "display": {
    "singular": "API Request",
    "plural": "API Requests"
  }
}
```

```json Response
{
  "id": "api_calls",
  "name": "API Requests",
  "type": "single_use",
  "consumable": true,
  "display": {
    "singular": "API Request",
    "plural": "API Requests"
  },
  "archived": false
}
```
