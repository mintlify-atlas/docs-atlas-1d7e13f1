---
title: 'Get Feature'
openapi: 'GET /features/{feature_id}'
---

Retrieve details about a specific feature by its ID.

## Path Parameters

<ParamField path="feature_id" type="string" required>
  The ID of the feature to retrieve
</ParamField>

## Response

<ResponseField name="id" type="string">
  The ID of the feature
</ResponseField>

<ResponseField name="name" type="string">
  The name of the feature
</ResponseField>

<ResponseField name="type" type="enum">
  The type of the feature: `single_use`, `continuous_use`, or `credit_system`
</ResponseField>

<ResponseField name="consumable" type="boolean">
  Whether the feature is consumable
</ResponseField>

<ResponseField name="event_names" type="array">
  Array of event names associated with this feature
</ResponseField>

<ResponseField name="credit_schema" type="array">
  Credit schema mapping for credit_system features
  
  <Expandable title="Schema Item">
    <ResponseField name="metered_feature_id" type="string">
      The ID of the mapped feature
    </ResponseField>
    
    <ResponseField name="credit_cost" type="number">
      The credit cost
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="display" type="object">
  Display names for the feature
  
  <Expandable title="Display">
    <ResponseField name="singular" type="string">
      Singular display name
    </ResponseField>
    
    <ResponseField name="plural" type="string">
      Plural display name
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="archived" type="boolean">
  Whether the feature is archived. Archived features are hidden from the dashboard and list features endpoint.
</ResponseField>

## Example

```json Response
{
  "id": "api_calls",
  "name": "API Calls",
  "type": "single_use",
  "consumable": true,
  "display": {
    "singular": "API Call",
    "plural": "API Calls"
  },
  "archived": false
}
```
